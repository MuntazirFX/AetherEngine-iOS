# build-arm64.yml
# Manual-triggered build only. Clean version tags.
# AetherEngine-iOS · Clean-room.

name: Build AetherEngine IPA

on:
  workflow_dispatch:
    inputs:
      version:
        description: 'Version (e.g. v0.1.0)'
        required: true
        type: string
      publish_release:
        description: 'Publish as GitHub Release?'
        required: true
        type: boolean
        default: false

permissions:
  contents: write

jobs:
  build-ipa:
    name: Build iOS IPA
    runs-on: macos-14
    timeout-minutes: 60

    steps:
      # ============================================================
      # 1. CHECKOUT
      # ============================================================
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 1

      - name: Show toolchain
        run: |
          xcodebuild -version
          xcrun --sdk iphoneos --show-sdk-path
          swift --version
          uname -m

      # ============================================================
      # 2. DEPENDENCIES
      # ============================================================
      - name: Install CMake
        run: |
          if ! command -v cmake >/dev/null 2>&1; then
            brew install cmake
          fi

      - name: Install XcodeGen
        run: |
          if ! command -v xcodegen >/dev/null 2>&1; then
            brew install xcodegen
          fi
          xcodegen --version

      - name: Make build scripts executable
        run: |
          chmod +x build/scripts/build_ios.sh 2>/dev/null || true
          chmod +x build/scripts/package_ipa.sh 2>/dev/null || true

      # ============================================================
      # 3. SANITY COMPILE
      # ============================================================
      - name: Sanity compile engine (host)
        run: |
          set -euo pipefail
          mkdir -p /tmp/sanity
          cd /tmp/sanity
          for f in \
            "$GITHUB_WORKSPACE/engine/core/AetherCore.c" \
            "$GITHUB_WORKSPACE/engine/core/AetherEngine.c" \
            "$GITHUB_WORKSPACE/engine/core/AetherMath.c" \
            "$GITHUB_WORKSPACE/engine/core/AetherArray.c" \
            "$GITHUB_WORKSPACE/engine/core/AetherEvent.c" \
            "$GITHUB_WORKSPACE/engine/game/AetherGameManager.c" \
            "$GITHUB_WORKSPACE/engine/game/AetherManifest.c" \
            "$GITHUB_WORKSPACE/engine/input/AetherInput.c" \
            "$GITHUB_WORKSPACE/engine/input/AetherTouch.c" \
            "$GITHUB_WORKSPACE/engine/config/AetherSettings.c" \
            "$GITHUB_WORKSPACE/engine/fs/AetherFS.c" \
            "$GITHUB_WORKSPACE/engine/fs/AetherPak.c" \
            "$GITHUB_WORKSPACE/engine/audio/AetherAudio.c" \
            "$GITHUB_WORKSPACE/engine/render/AetherRender.c" \
            "$GITHUB_WORKSPACE/engine/bsp/AetherBSP.c" \
            "$GITHUB_WORKSPACE/engine/bsp/AetherBSPGeometry.c" \
            "$GITHUB_WORKSPACE/engine/player/AetherPlayer.c" \
            "$GITHUB_WORKSPACE/engine/player/AetherCollision.c" \
            "$GITHUB_WORKSPACE/engine/texture/AetherTexture.c" \
            "$GITHUB_WORKSPACE/engine/model/AetherMDL.c" \
            "$GITHUB_WORKSPACE/engine/model/AetherMDLGeometry.c" \
            "$GITHUB_WORKSPACE/engine/entity/AetherEntity.c" \
            "$GITHUB_WORKSPACE/engine/entity/AetherEntityBase.c" \
          ; do
            if [ -f "$f" ]; then
              echo "→ cc -c $(basename $f)"
              cc -std=c11 -Wall -Wextra -Wno-unused-parameter \
                -I"$GITHUB_WORKSPACE/engine/core" \
                -I"$GITHUB_WORKSPACE/engine/game" \
                -I"$GITHUB_WORKSPACE/engine/input" \
                -I"$GITHUB_WORKSPACE/engine/config" \
                -I"$GITHUB_WORKSPACE/engine/fs" \
                -I"$GITHUB_WORKSPACE/engine/audio" \
                -I"$GITHUB_WORKSPACE/engine/render" \
                -I"$GITHUB_WORKSPACE/engine/bsp" \
                -I"$GITHUB_WORKSPACE/engine/player" \
                -I"$GITHUB_WORKSPACE/engine/texture" \
                -I"$GITHUB_WORKSPACE/engine/model" \
                -I"$GITHUB_WORKSPACE/engine/entity" \
                -c "$f" -o "$(basename "$f" .c).o"
            fi
          done
          echo "✔ Sanity compile passed"

      # ============================================================
      # 4. BUILD IPA
      # ============================================================
      - name: Build iOS app (unsigned)
        run: bash build/scripts/build_ios.sh

      - name: Package unsigned IPA
        run: bash build/scripts/package_ipa.sh

      - name: Verify IPA
        run: |
          set -euo pipefail
          IPA_PATH="build/out/AetherEngine.ipa"
          test -f "$IPA_PATH" || (echo "IPA missing!" && exit 1)
          echo "IPA size:"
          ls -lh "$IPA_PATH"

      # ============================================================
      # 5. METADATA
      # ============================================================
      - name: Compute IPA metadata
        id: meta
        run: |
          IPA=build/out/AetherEngine.ipa
          SIZE=$(du -sh "$IPA" | cut -f1)
          SHA=$(shasum -a 256 "$IPA" | awk '{print $1}')
          SHORT_SHA=$(echo "${{ github.sha }}" | cut -c1-7)
          DATE=$(date -u +"%Y-%m-%d %H:%M UTC")
          VERSION="${{ github.event.inputs.version }}"
          echo "size=$SIZE"       >> $GITHUB_OUTPUT
          echo "sha=$SHA"         >> $GITHUB_OUTPUT
          echo "short=$SHORT_SHA" >> $GITHUB_OUTPUT
          echo "date=$DATE"       >> $GITHUB_OUTPUT
          echo "version=$VERSION" >> $GITHUB_OUTPUT

      # ============================================================
      # 6. UPLOAD ARTIFACT (IPA only, no static lib)
      # ============================================================
      - name: Upload IPA artifact
        uses: actions/upload-artifact@v4
        with:
          name: AetherEngine-${{ github.event.inputs.version }}
          path: build/out/AetherEngine.ipa
          retention-days: 30
          if-no-files-found: error

      # ============================================================
      # 7. CREATE RELEASE (only if requested)
      # ============================================================
      - name: Build release notes
        if: ${{ github.event.inputs.publish_release == 'true' }}
        run: |
          cat > /tmp/notes.md <<EOF
          ## AetherEngine-iOS — ${{ github.event.inputs.version }}

          **Built:** ${{ steps.meta.outputs.date }}

          ### 📦 Download
          - \`AetherEngine.ipa\` — sideload via Sideloadly / AltStore / TrollStore

          ### 📋 Metadata
          - **Size:** ${{ steps.meta.outputs.size }}
          - **SHA256:** \`${{ steps.meta.outputs.sha }}\`
          - **Commit:** \`${{ github.sha }}\` (\`${{ steps.meta.outputs.short }}\`)

          ### 🎮 Supported Games
          | Game | Folder |
          |------|--------|
          | Half-Life | \`valve\` |
          | Half-Life: Blue Shift | \`bshift\` |
          | Half-Life: Opposing Force | \`gearbox\` |
          | Counter-Strike 1.6 | \`cstrike\` |
          | Counter-Strike: Condition Zero | \`czero\` |

          ### ⚠️ Legal
          No copyrighted assets bundled. User must provide their own
          legally-obtained game data in \`Documents/<game_folder>/\`.

          ### 🔧 Install
          1. Download \`AetherEngine.ipa\`
          2. Use Sideloadly / AltStore / TrollStore
          3. Open the app — it auto-creates the 5 game folders in Files app
          4. Drop your legally-owned game data into the correct folder
          EOF

      - name: Create GitHub Release
        if: ${{ github.event.inputs.publish_release == 'true' }}
        uses: softprops/action-gh-release@v2
        with:
          tag_name: ${{ github.event.inputs.version }}
          name: "AetherEngine ${{ github.event.inputs.version }}"
          body_path: /tmp/notes.md
          draft: false
          prerelease: false
          make_latest: true
          files: |
            build/out/AetherEngine.ipa
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

      # ============================================================
      # 8. SUMMARY
      # ============================================================
      - name: Summary
        if: always()
        run: |
          echo "### AetherEngine Build Summary" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY
          echo "- **Version:** ${{ github.event.inputs.version }}" >> $GITHUB_STEP_SUMMARY
          echo "- **Commit:** \`${{ github.sha }}\`" >> $GITHUB_STEP_SUMMARY
          echo "- **Runner:** macOS 14" >> $GITHUB_STEP_SUMMARY
          echo "- **Target:** iOS 14+ ARM64" >> $GITHUB_STEP_SUMMARY
          echo "- **Release:** ${{ github.event.inputs.publish_release }}" >> $GITHUB_STEP_SUMMARY
          if [ -f build/out/AetherEngine.ipa ]; then
            echo "- **IPA size:** $(du -sh build/out/AetherEngine.ipa | cut -f1)" >> $GITHUB_STEP_SUMMARY
            echo "- **Status:** ✅ Build succeeded" >> $GITHUB_STEP_SUMMARY
          else
            echo "- **Status:** ❌ Build failed" >> $GITHUB_STEP_SUMMARY
          fi
